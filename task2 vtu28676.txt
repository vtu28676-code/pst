#include <stdio.h>

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int index;

    printf("Enter the index: ");
    scanf("%d", &index);

    if (index >= 0 && index < 5) {
        printf("Element at index %d is %d\n", index, arr[index]);
    } else {
        printf("Invalid index!\n");
    }

    return 0;
}
