import java.util.*;

public class Solution {
    public List<String> findAndReplacePattern(String[] words, String pattern) {
        List<String> result = new ArrayList<>();
        
        for (String word : words) {
            if (matches(word, pattern)) {
                result.add(word);
            }
        }
        
        return result;
    }

    private boolean matches(String word, String pattern) {
        if (word.length() != pattern.length()) {
            return false;
        }

        // Map characters from pattern -> word and word -> pattern to enforce bijection
        Map<Character, Character> pToW = new HashMap<>();
        Map<Character, Character> wToP = new HashMap<>();

        for (int i = 0; i < word.length(); i++) {
            char p = pattern.charAt(i);
            char w = word.charAt(i);

            if (!pToW.containsKey(p)) {
                pToW.put(p, w);
            }
            if (!wToP.containsKey(w)) {
                wToP.put(w, p);
            }

            // Check if both mappings match
            if (pToW.get(p) != w || wToP.get(w) != p) {
                return false;
            }
        }

        return true;
    }
}
