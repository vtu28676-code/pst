import java.io.*;
import java.util.*;

public class Solution {

    public static void matrixRotation(List<List<Integer>> matrix, int r) {
        int m = matrix.size();
        int n = matrix.get(0).size();
        int numLayers = Math.min(m, n) / 2;

        for (int layer = 0; layer < numLayers; layer++) {
            List<Integer> layerElements = new ArrayList<>();

          
            for (int j = layer; j < n - layer; j++) {
                layerElements.add(matrix.get(layer).get(j));
            }
        
            for (int i = layer + 1; i < m - layer - 1; i++) {
                layerElements.add(matrix.get(i).get(n - 1 - layer));
            }
           
            for (int j = n - 1 - layer; j >= layer; j--) {
                layerElements.add(matrix.get(m - 1 - layer).get(j));
            }
           
            for (int i = m - 2 - layer; i > layer; i--) {
                layerElements.add(matrix.get(i).get(layer));
            }

          
            int L = layerElements.size();
            int effR = r % L;

          \
            int idx = 0;

         
            for (int j = layer; j < n - layer; j++) {
                matrix.get(layer).set(j, layerElements.get((idx + effR) % L));
                idx++;
            }
            // Right column
            for (int i = layer + 1; i < m - layer - 1; i++) {
                matrix.get(i).set(n - 1 - layer, layerElements.get((idx + effR) % L));
                idx++;
            }
            // Bottom row
            for (int j = n - 1 - layer; j >= layer; j--) {
                matrix.get(m - 1 - layer).set(j, layerElements.get((idx + effR) % L));
                idx++;
            }
            // Left column
            for (int i = m - 2 - layer; i > layer; i--) {
                matrix.get(i).set(layer, layerElements.get((idx + effR) % L));
                idx++;
            }
        }

        // 4. Print the resulting matrix
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                sb.append(matrix.get(i).get(j)).append(j == n - 1 ? "" : " ");
            }
            sb.append("\n");
        }
        System.out.print(sb.toString());
    }

    public static void main(String[] args) throws IOException {
        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(System.in));
        String[] firstMultipleInput = bufferedReader.readLine().replaceAll("\\s+$", "").split(" ");

        int m = Integer.parseInt(firstMultipleInput[0]);
        int n = Integer.parseInt(firstMultipleInput[1]);
        int r = Integer.parseInt(firstMultipleInput[2]);

        List<List<Integer>> matrix = new ArrayList<>();

        for (int i = 0; i < m; i++) {
            String[] matrixRowTempItems = bufferedReader.readLine().replaceAll("\\s+$", "").split(" ");
            List<Integer> matrixRowItems = new ArrayList<>();

            for (int j = 0; j < n; j++) {
                int matrixItem = Integer.parseInt(matrixRowTempItems[j]);
                matrixRowItems.add(matrixItem);
            }

            matrix.add(matrixRowItems);
        }

        matrixRotation(matrix, r);
        bufferedReader.close();
    }
}
